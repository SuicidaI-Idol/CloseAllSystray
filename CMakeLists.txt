cmake_minimum_required(VERSION 3.20)
project(CloseAllSystray LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This application requires Windows.")
endif()
enable_language(RC)
add_executable(CloseAllSystray WIN32 src/main.cpp src/app.rc)
target_include_directories(CloseAllSystray PRIVATE src)
target_compile_features(CloseAllSystray PRIVATE cxx_std_17)
set_target_properties(CloseAllSystray PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
target_compile_definitions(CloseAllSystray PRIVATE _WIN32_WINNT=0x0A00)
target_link_libraries(CloseAllSystray PRIVATE comctl32 ole32 shell32 uuid advapi32 gdi32 dwmapi)
if(MINGW)
  target_compile_options(CloseAllSystray PRIVATE -Wall -Wextra -Wpedantic -finput-charset=UTF-8 -fexec-charset=UTF-8)
  target_link_options(CloseAllSystray PRIVATE -municode -static)
elseif(MSVC)
  target_compile_options(CloseAllSystray PRIVATE /W4 /utf-8)
endif()
